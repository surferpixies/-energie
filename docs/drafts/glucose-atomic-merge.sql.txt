-- DRAFT ONLY — do not execute in production before validation.
-- Review PostgreSQL 16+ availability and verify with integration tests before use.
-- Makes glucose arrays and deletion markers merge atomically during daily_logs UPDATE.
-- PostgreSQL row locking serializes concurrent UPDATE/UPSERT writes to the same day.
-- This does not change professional access/RLS and does not edit other supplements.
CREATE OR REPLACE FUNCTION public.merge_glucose_supplements(
  previous jsonb, incoming jsonb
) RETURNS jsonb
LANGUAGE plpgsql IMMUTABLE
SET search_path = ''
AS $$
DECLARE
  original jsonb := CASE WHEN jsonb_typeof(previous) = 'object' THEN previous ELSE '{}'::jsonb END;
  requested jsonb := CASE WHEN jsonb_typeof(incoming) = 'object' THEN incoming ELSE '{}'::jsonb END;
  deleted jsonb := '{}'::jsonb;
  records jsonb := '{}'::jsonb;
  result_array jsonb := '[]'::jsonb;
  entry jsonb;
  item_id text;
  deletion_id text;
  deletion_time text;
  time_old timestamptz;
  time_new timestamptz;
BEGIN
  FOR deletion_id, deletion_time IN
    SELECT key, value #>> '{}' FROM pg_catalog.jsonb_each(
      CASE WHEN jsonb_typeof(original -> 'glucoseTombstones') = 'object'
        THEN original -> 'glucoseTombstones' ELSE '{}'::jsonb END)
  LOOP
    IF deletion_id <> '' AND pg_catalog.pg_input_is_valid(deletion_time, 'timestamp with time zone') THEN
      deleted := pg_catalog.jsonb_set(deleted, ARRAY[deletion_id], pg_catalog.to_jsonb(deletion_time), true);
    END IF;
  END LOOP;
  FOR deletion_id, deletion_time IN
    SELECT key, value #>> '{}' FROM pg_catalog.jsonb_each(
      CASE WHEN jsonb_typeof(requested -> 'glucoseTombstones') = 'object'
        THEN requested -> 'glucoseTombstones' ELSE '{}'::jsonb END)
  LOOP
    IF deletion_id <> '' AND pg_catalog.pg_input_is_valid(deletion_time, 'timestamp with time zone') THEN
      IF NOT (deleted ? deletion_id)
          OR deletion_time::timestamptz > (deleted ->> deletion_id)::timestamptz THEN
        deleted := pg_catalog.jsonb_set(deleted, ARRAY[deletion_id], pg_catalog.to_jsonb(deletion_time), true);
      END IF;
    END IF;
  END LOOP;

  -- Old first, new last: on equal/unversioned timestamps, the incoming entry wins.
  FOR entry IN
    SELECT value FROM (
      SELECT 0 AS source_order, ordinality AS row_order, value
      FROM pg_catalog.jsonb_array_elements(
      CASE WHEN jsonb_typeof(original -> 'glucoseMeasurements') = 'array'
        THEN original -> 'glucoseMeasurements' ELSE '[]'::jsonb END) WITH ORDINALITY
      UNION ALL
      SELECT 1 AS source_order, ordinality AS row_order, value
      FROM pg_catalog.jsonb_array_elements(
      CASE WHEN jsonb_typeof(requested -> 'glucoseMeasurements') = 'array'
        THEN requested -> 'glucoseMeasurements' ELSE '[]'::jsonb END) WITH ORDINALITY
    ) ordered_entries
    ORDER BY source_order, row_order
  LOOP
    item_id := entry ->> 'id';
    IF item_id IS NULL OR item_id = '' THEN CONTINUE; END IF;
    IF records ? item_id THEN
      time_old := CASE WHEN pg_catalog.pg_input_is_valid(records -> item_id ->> 'updatedAt', 'timestamp with time zone')
        THEN (records -> item_id ->> 'updatedAt')::timestamptz ELSE '-infinity'::timestamptz END;
      time_new := CASE WHEN pg_catalog.pg_input_is_valid(entry ->> 'updatedAt', 'timestamp with time zone')
        THEN (entry ->> 'updatedAt')::timestamptz ELSE '-infinity'::timestamptz END;
      IF time_new < time_old THEN CONTINUE; END IF;
    END IF;
    records := pg_catalog.jsonb_set(records, ARRAY[item_id], entry, true);
  END LOOP;

  FOR item_id, entry IN SELECT key, value FROM pg_catalog.jsonb_each(records)
  LOOP
    IF deleted ? item_id THEN
      time_new := CASE WHEN pg_catalog.pg_input_is_valid(entry ->> 'updatedAt', 'timestamp with time zone')
        THEN (entry ->> 'updatedAt')::timestamptz ELSE '-infinity'::timestamptz END;
      IF (deleted ->> item_id)::timestamptz >= time_new THEN CONTINUE; END IF;
    END IF;
    result_array := result_array || pg_catalog.jsonb_build_array(entry);
  END LOOP;
  RETURN requested || pg_catalog.jsonb_build_object(
    'glucoseMeasurements', result_array,
    'glucoseTombstones', deleted
  );
END;
$$;

-- WARNING: migration requires PostgreSQL version supporting pg_input_is_valid
-- (PostgreSQL 16+). Check server_version_num before applying.
CREATE OR REPLACE FUNCTION public.preserve_daily_glucose()
RETURNS trigger
LANGUAGE plpgsql
SET search_path = ''
AS $$
BEGIN
  IF NEW.supplements IS NOT NULL THEN
    NEW.supplements := public.merge_glucose_supplements(OLD.supplements, NEW.supplements);
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS preserve_daily_glucose_trigger ON public.daily_logs;
CREATE TRIGGER preserve_daily_glucose_trigger
  BEFORE UPDATE ON public.daily_logs
  FOR EACH ROW EXECUTE FUNCTION public.preserve_daily_glucose();
